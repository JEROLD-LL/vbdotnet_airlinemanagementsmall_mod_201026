Public Class BookingRepository
    Public Shared Function GetByUserId(userId As Integer) As List(Of Booking)
        Dim bookings As New List(Of Booking)
        Dim query As String = "SELECT b.id, b.flight_id, b.user_id, b.passenger_name, b.passenger_email, b.seat_number, b.booking_date FROM Booking b WHERE b.user_id = @user_id ORDER BY b.booking_date DESC"
        Using reader As NpgsqlDataReader = DatabaseHelper.ExecuteReader(query, New NpgsqlParameter("@user_id", userId))
            While reader.Read()
                bookings.Add(New Booking With {
                    .Id = reader.GetInt32(0),
                    .FlightId = reader.GetInt32(1),
                    .UserId = If(reader.IsDBNull(2), Nothing, New Integer?(reader.GetInt32(2))),
                    .PassengerName = reader.GetString(3),
                    .PassengerEmail = reader.GetString(4),
                    .SeatNumber = If(reader.IsDBNull(5), String.Empty, reader.GetString(5)),
                    .BookingDate = reader.GetDateTime(6)
                })
            End While
        End Using
        Return bookings
    End Function

    Public Shared Function GetAll() As List(Of Booking)
        Dim bookings As New List(Of Booking)
        Dim query As String = "SELECT b.id, b.flight_id, b.user_id, b.passenger_name, b.passenger_email, b.seat_number, b.booking_date FROM Booking b ORDER BY b.booking_date DESC"
        Using reader As NpgsqlDataReader = DatabaseHelper.ExecuteReader(query)
            While reader.Read()
                bookings.Add(New Booking With {
                    .Id = reader.GetInt32(0),
                    .FlightId = reader.GetInt32(1),
                    .UserId = If(reader.IsDBNull(2), Nothing, New Integer?(reader.GetInt32(2))),
                    .PassengerName = reader.GetString(3),
                    .PassengerEmail = reader.GetString(4),
                    .SeatNumber = If(reader.IsDBNull(5), String.Empty, reader.GetString(5)),
                    .BookingDate = reader.GetDateTime(6)
                })
            End While
        End Using
        Return bookings
    End Function

    Public Shared Function GetById(id As Integer) As Booking
        Dim query As String = "SELECT b.id, b.flight_id, b.user_id, b.passenger_name, b.passenger_email, b.seat_number, b.booking_date FROM Booking b WHERE b.id = @id"
        Using reader As NpgsqlDataReader = DatabaseHelper.ExecuteReader(query, New NpgsqlParameter("@id", id))
            If reader.Read() Then
                Return New Booking With {
                    .Id = reader.GetInt32(0),
                    .FlightId = reader.GetInt32(1),
                    .UserId = If(reader.IsDBNull(2), Nothing, New Integer?(reader.GetInt32(2))),
                    .PassengerName = reader.GetString(3),
                    .PassengerEmail = reader.GetString(4),
                    .SeatNumber = If(reader.IsDBNull(5), String.Empty, reader.GetString(5)),
                    .BookingDate = reader.GetDateTime(6)
                }
            End If
        End Using
        Return Nothing
    End Function

    Public Shared Function Create(booking As Booking, transaction As NpgsqlTransaction) As Integer
        Dim query As String = "INSERT INTO Booking (flight_id, user_id, passenger_name, passenger_email, seat_number, booking_date) VALUES (@flight_id, @user_id, @passenger_name, @passenger_email, @seat_number, @booking_date) RETURNING id"
        Using cmd As New NpgsqlCommand(query, transaction.Connection, transaction)
            cmd.Parameters.Add(New NpgsqlParameter("@flight_id", booking.FlightId))
            If booking.UserId.HasValue Then
                cmd.Parameters.Add(New NpgsqlParameter("@user_id", booking.UserId.Value))
            Else
                cmd.Parameters.Add(New NpgsqlParameter("@user_id", DBNull.Value))
            End If
            cmd.Parameters.Add(New NpgsqlParameter("@passenger_name", booking.PassengerName))
            cmd.Parameters.Add(New NpgsqlParameter("@passenger_email", booking.PassengerEmail))
            If Not String.IsNullOrEmpty(booking.SeatNumber) Then
                cmd.Parameters.Add(New NpgsqlParameter("@seat_number", booking.SeatNumber))
            Else
                cmd.Parameters.Add(New NpgsqlParameter("@seat_number", DBNull.Value))
            End If
            cmd.Parameters.Add(New NpgsqlParameter("@booking_date", booking.BookingDate))
            Return Convert.ToInt32(cmd.ExecuteScalar())
        End Using
    End Function

    Public Shared Sub Update(booking As Booking)
        Dim query As String = "UPDATE Booking SET passenger_name = @passenger_name, passenger_email = @passenger_email, seat_number = @seat_number WHERE id = @id"
        Dim parameters As New List(Of NpgsqlParameter) From {
            New NpgsqlParameter("@passenger_name", booking.PassengerName),
            New NpgsqlParameter("@passenger_email", booking.PassengerEmail),
            New NpgsqlParameter("@id", booking.Id)
        }
        If Not String.IsNullOrEmpty(booking.SeatNumber) Then
            parameters.Add(New NpgsqlParameter("@seat_number", booking.SeatNumber))
        Else
            parameters.Add(New NpgsqlParameter("@seat_number", DBNull.Value))
        End If
        DatabaseHelper.ExecuteNonQuery(query, parameters.ToArray())
    End Sub

    Public Shared Sub Delete(id As Integer, transaction As NpgsqlTransaction)
        Dim query As String = "DELETE FROM Booking WHERE id = @id"
        Using cmd As New NpgsqlCommand(query, transaction.Connection, transaction)
            cmd.Parameters.Add(New NpgsqlParameter("@id", id))
            cmd.ExecuteNonQuery()
        End Using
    End Sub
End Class
