Public Class FlightsForm
    Inherits Form

    Private dgvFlights As DataGridView
    Private btnRefresh As Button
    Private btnView As Button
    Private btnBook As Button
    Private btnEdit As Button
    Private btnDelete As Button

    Public Sub New()
        InitializeComponent()
        LoadFlights()
    End Sub

    Private Sub InitializeComponent()
        Me.Text = "Available Flights"
        Me.Size = New Size(1000, 500)
        Me.StartPosition = FormStartPosition.CenterScreen

        ' Data Grid View
        dgvFlights = New DataGridView()
        dgvFlights.Location = New Point(10, 10)
        dgvFlights.Size = New Size(960, 400)
        dgvFlights.AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill
        dgvFlights.SelectionMode = DataGridViewSelectionMode.FullRowSelect
        dgvFlights.ReadOnly = True
        dgvFlights.AllowUserToAddRows = False
        Me.Controls.Add(dgvFlights)

        ' Buttons
        btnRefresh = New Button()
        btnRefresh.Text = "Refresh"
        btnRefresh.Location = New Point(10, 420)
        btnRefresh.Size = New Size(100, 30)
        AddHandler btnRefresh.Click, AddressOf BtnRefresh_Click
        Me.Controls.Add(btnRefresh)

        btnView = New Button()
        btnView.Text = "View Details"
        btnView.Location = New Point(120, 420)
        btnView.Size = New Size(100, 30)
        AddHandler btnView.Click, AddressOf BtnView_Click
        Me.Controls.Add(btnView)

        btnBook = New Button()
        btnBook.Text = "Book Flight"
        btnBook.Location = New Point(230, 420)
        btnBook.Size = New Size(100, 30)
        AddHandler btnBook.Click, AddressOf BtnBook_Click
        Me.Controls.Add(btnBook)

        If SessionManager.IsAdmin() Then
            btnEdit = New Button()
            btnEdit.Text = "Edit"
            btnEdit.Location = New Point(340, 420)
            btnEdit.Size = New Size(100, 30)
            AddHandler btnEdit.Click, AddressOf BtnEdit_Click
            Me.Controls.Add(btnEdit)

            btnDelete = New Button()
            btnDelete.Text = "Delete"
            btnDelete.Location = New Point(450, 420)
            btnDelete.Size = New Size(100, 30)
            AddHandler btnDelete.Click, AddressOf BtnDelete_Click
            Me.Controls.Add(btnDelete)
        End If
    End Sub

    Private Sub LoadFlights()
        Dim flights As List(Of Flight) = FlightRepository.GetAll()

        dgvFlights.Columns.Clear()
        dgvFlights.Columns.Add("Id", "ID")
        dgvFlights.Columns("Id").Visible = False
        dgvFlights.Columns.Add("FlightNumber", "Flight Number")
        dgvFlights.Columns.Add("Origin", "Origin")
        dgvFlights.Columns.Add("Destination", "Destination")
        dgvFlights.Columns.Add("DepartureDate", "Departure Date")
        dgvFlights.Columns.Add("DepartureTime", "Departure Time")
        dgvFlights.Columns.Add("ArrivalTime", "Arrival Time")
        dgvFlights.Columns.Add("Price", "Price")
        dgvFlights.Columns("Price").DefaultCellStyle.Format = "C"
        dgvFlights.Columns.Add("AvailableSeats", "Available Seats")

        dgvFlights.Rows.Clear()
        For Each flight In flights
            dgvFlights.Rows.Add(flight.Id, flight.FlightNumber, flight.Origin, flight.Destination,
                              flight.DepartureDate.ToString("yyyy-MM-dd"),
                              flight.DepartureTime.ToString("hh\:mm"),
                              flight.ArrivalTime.ToString("hh\:mm"),
                              flight.Price, flight.AvailableSeats)
        Next
    End Sub

    Private Function GetSelectedFlightId() As Integer?
        If dgvFlights.SelectedRows.Count > 0 Then
            Return Convert.ToInt32(dgvFlights.SelectedRows(0).Cells("Id").Value)
        End If
        Return Nothing
    End Function

    Private Sub BtnRefresh_Click(sender As Object, e As EventArgs)
        LoadFlights()
    End Sub

    Private Sub BtnView_Click(sender As Object, e As EventArgs)
        Dim flightId As Integer? = GetSelectedFlightId()
        If flightId.HasValue Then
            Dim viewForm As New ViewFlightForm(flightId.Value)
            viewForm.ShowDialog()
            LoadFlights()
        Else
            MessageBox.Show("Please select a flight.", "No Selection", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        End If
    End Sub

    Private Sub BtnBook_Click(sender As Object, e As EventArgs)
        Dim flightId As Integer? = GetSelectedFlightId()
        If flightId.HasValue Then
            Dim bookForm As New BookFlightForm(flightId.Value)
            bookForm.ShowDialog()
            LoadFlights()
        Else
            MessageBox.Show("Please select a flight.", "No Selection", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        End If
    End Sub

    Private Sub BtnEdit_Click(sender As Object, e As EventArgs)
        Dim flightId As Integer? = GetSelectedFlightId()
        If flightId.HasValue Then
            Dim editForm As New EditFlightForm(flightId.Value)
            editForm.ShowDialog()
            LoadFlights()
        Else
            MessageBox.Show("Please select a flight.", "No Selection", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        End If
    End Sub

    Private Sub BtnDelete_Click(sender As Object, e As EventArgs)
        Dim flightId As Integer? = GetSelectedFlightId()
        If flightId.HasValue Then
            If MessageBox.Show("Are you sure you want to delete this flight?", "Confirm Delete", MessageBoxButtons.YesNo, MessageBoxIcon.Question) = DialogResult.Yes Then
                Try
                    FlightRepository.Delete(flightId.Value)
                    MessageBox.Show("Flight deleted successfully!", "Success", MessageBoxButtons.OK, MessageBoxIcon.Information)
                    LoadFlights()
                Catch ex As Exception
                    MessageBox.Show("Error deleting flight: " & ex.Message, "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
                End Try
            End If
        Else
            MessageBox.Show("Please select a flight.", "No Selection", MessageBoxButtons.OK, MessageBoxIcon.Warning)
        End If
    End Sub
End Class
