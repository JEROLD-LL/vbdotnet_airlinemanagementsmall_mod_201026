Public Shared Function ValidateUsername(username As String) As List(Of String)
        Dim errors As New List(Of String)
        If String.IsNullOrWhiteSpace(username) Then
            errors.Add("Username is required.")
        ElseIf username.Length < 3 OrElse username.Length > 50 Then
            errors.Add("Username must be between 3 and 50 characters.")
        ElseIf Not Regex.IsMatch(username, "^[a-zA-Z0-9_]+$") Then
            errors.Add("Username can only contain letters, numbers, and underscores.")
        End If
        Return errors
    End Function

    Private Sub BtnRegister_Click(sender As Object, e As EventArgs)
        lblError.Text = ""

        Dim errors As New List(Of String)

        Dim username As String = txtUsername.Text.Trim()
        Dim email As String = txtEmail.Text.Trim()
        Dim fullName As String = txtFullName.Text.Trim()
        Dim password As String = txtPassword.Text

        ' Validate username
        errors.AddRange(ValidationHelper.ValidateUsername(username))
        If errors.Count = 0 AndAlso UserRepository.UsernameExists(username) Then
            errors.Add("Username already exists.")
        End If

        ' Validate email
        errors.AddRange(ValidationHelper.ValidateEmail(email))
        If errors.Count = 0 AndAlso UserRepository.EmailExists(email) Then
            errors.Add("Email already exists.")
        End If

        ' Validate full name
        errors.AddRange(ValidationHelper.ValidateFullName(fullName))

        ' Validate password
        errors.AddRange(ValidationHelper.ValidatePassword(password))

        If errors.Count > 0 Then
            lblError.Text = String.Join(vbCrLf, errors)
            Return
        End If

        ' Create user
        Dim user As New User With {
            .Username = username,
            .Password = UserRepository.HashPassword(password),
            .Email = email,
            .FullName = fullName,
            .Role = "user",
            .CreatedAt = DateTime.Now
        }

        Try
            UserRepository.Create(user)
            MessageBox.Show("Registration successful! Please login.", "Success", MessageBoxButtons.OK, MessageBoxIcon.Information)
            Me.Hide()
            Dim loginForm As New LoginForm()
            loginForm.Show()
        Catch ex As Exception
            lblError.Text = "An error occurred: " & ex.Message
        End Try
    End Sub

    Private Sub BtnCancel_Click(sender As Object, e As EventArgs)
        Me.Hide()
        Dim loginForm As New LoginForm()
        loginForm.Show()
    End Sub

    Private Sub LoadProfile()
        If Not SessionManager.IsAuthenticated() Then
            MessageBox.Show("Please login first.", "Not Authenticated", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Me.Close()
            Return
        End If

        user = UserRepository.GetById(SessionManager.CurrentUser.Id)
        If user Is Nothing Then
            MessageBox.Show("User not found.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
            Me.Close()
            Return
        End If

        txtEmail.Text = user.Email
        txtFullName.Text = user.FullName
    End Sub

    Private Sub BtnUpdate_Click(sender As Object, e As EventArgs)
        lblError.Text = ""

        Dim errors As New List(Of String)

        Dim email As String = txtEmail.Text.Trim()
        Dim fullName As String = txtFullName.Text.Trim()
        Dim password As String = txtPassword.Text

        errors.AddRange(ValidationHelper.ValidateEmail(email))
        errors.AddRange(ValidationHelper.ValidateFullName(fullName))

        If Not String.IsNullOrWhiteSpace(password) Then
            errors.AddRange(ValidationHelper.ValidatePassword(password))
        End If

        If errors.Count > 0 Then
            lblError.Text = String.Join(vbCrLf, errors)
            Return
        End If

        user.Email = email
        user.FullName = fullName

        Try
            UserRepository.Update(user)

            If Not String.IsNullOrWhiteSpace(password) Then
                UserRepository.UpdatePassword(user.Id, UserRepository.HashPassword(password))
            End If

            SessionManager.CurrentUser.FullName = user.FullName
            SessionManager.CurrentUser.Email = user.Email

            MessageBox.Show("Profile updated successfully!", "Success", MessageBoxButtons.OK, MessageBoxIcon.Information)
            Me.Close()
        Catch ex As Exception
            lblError.Text = "An error occurred: " & ex.Message
        End Try
    End Sub

    Private Sub BtnCancel_Click(sender As Object, e As EventArgs)
        Me.Close()
    End Sub
End Class

        Dim hash As String = HashPassword(plainPassword)

    Public Shared Function HashPassword(password As String) As String
        Using sha1 As SHA1 = SHA1.Create()
            Dim hash As Byte() = sha1.ComputeHash(Encoding.UTF8.GetBytes(password))
            Return BitConverter.ToString(hash).Replace("-", "").ToUpper()
        End Using
    End Function

Public Shared Function GetByUserId(userId As Integer) As List(Of Booking)
        Dim bookings As New List(Of Booking)
        Dim query As String = "SELECT b.id, b.flight_id, b.user_id, b.passenger_name, b.passenger_email, b.seat_number, b.booking_date FROM Booking b WHERE b.user_id = @user_id ORDER BY b.booking_date DESC"
        Using reader As NpgsqlDataReader = DatabaseHelper.ExecuteReader(query, New NpgsqlParameter("@user_id", userId))
            While reader.Read()
                bookings.Add(New Booking With {
                    .Id = reader.GetInt32(0),
                    .FlightId = reader.GetInt32(1),
                    .UserId = If(reader.IsDBNull(2), Nothing, New Integer?(reader.GetInt32(2))),
                    .PassengerName = reader.GetString(3),
                    .PassengerEmail = reader.GetString(4),
                    .SeatNumber = If(reader.IsDBNull(5), String.Empty, reader.GetString(5)),
                    .BookingDate = reader.GetDateTime(6)
                })
            End While
        End Using
        Return bookings
    End Function
