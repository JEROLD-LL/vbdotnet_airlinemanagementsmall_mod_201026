Public Class EditProfileForm
    Inherits Form

    Private user As User
    Private lblEmail As Label
    Private lblFullName As Label
    Private lblPassword As Label
    Private txtEmail As TextBox
    Private txtFullName As TextBox
    Private txtPassword As TextBox
    Private btnUpdate As Button
    Private btnCancel As Button
    Private lblError As Label

    Public Sub New()
        InitializeComponent()
        LoadProfile()
    End Sub

    Private Sub InitializeComponent()
        Me.Text = "Edit Profile"
        Me.Size = New Size(500, 350)
        Me.StartPosition = FormStartPosition.CenterScreen
        Me.FormBorderStyle = FormBorderStyle.FixedDialog
        Me.MaximizeBox = False

        Dim yPos As Integer = 20

        ' Email
        lblEmail = New Label()
        lblEmail.Text = "Email: *"
        lblEmail.AutoSize = True
        lblEmail.Location = New Point(20, yPos)
        Me.Controls.Add(lblEmail)

        txtEmail = New TextBox()
        txtEmail.Location = New Point(20, yPos + 20)
        txtEmail.Size = New Size(450, 25)
        Me.Controls.Add(txtEmail)
        yPos += 50

        ' Full Name
        lblFullName = New Label()
        lblFullName.Text = "Full Name: *"
        lblFullName.AutoSize = True
        lblFullName.Location = New Point(20, yPos)
        Me.Controls.Add(lblFullName)

        txtFullName = New TextBox()
        txtFullName.Location = New Point(20, yPos + 20)
        txtFullName.Size = New Size(450, 25)
        Me.Controls.Add(txtFullName)
        yPos += 50

        ' Password
        lblPassword = New Label()
        lblPassword.Text = "New Password (Optional):"
        lblPassword.AutoSize = True
        lblPassword.Location = New Point(20, yPos)
        Me.Controls.Add(lblPassword)

        txtPassword = New TextBox()
        txtPassword.Location = New Point(20, yPos + 20)
        txtPassword.Size = New Size(450, 25)
        txtPassword.PasswordChar = "*"c
        Me.Controls.Add(txtPassword)
        yPos += 50

        ' Error label
        lblError = New Label()
        lblError.ForeColor = Color.Red
        lblError.AutoSize = True
        lblError.Location = New Point(20, yPos)
        lblError.Size = New Size(450, 40)
        Me.Controls.Add(lblError)
        yPos += 50

        ' Buttons
        btnUpdate = New Button()
        btnUpdate.Text = "Update Profile"
        btnUpdate.Location = New Point(20, yPos)
        btnUpdate.Size = New Size(150, 30)
        AddHandler btnUpdate.Click, AddressOf BtnUpdate_Click
        Me.Controls.Add(btnUpdate)

        btnCancel = New Button()
        btnCancel.Text = "Cancel"
        btnCancel.Location = New Point(320, yPos)
        btnCancel.Size = New Size(150, 30)
        AddHandler btnCancel.Click, AddressOf BtnCancel_Click
        Me.Controls.Add(btnCancel)

        Me.AcceptButton = btnUpdate
    End Sub

    Private Sub LoadProfile()
        If Not SessionManager.IsAuthenticated() Then
            MessageBox.Show("Please login first.", "Not Authenticated", MessageBoxButtons.OK, MessageBoxIcon.Warning)
            Me.Close()
            Return
        End If

        user = UserRepository.GetById(SessionManager.CurrentUser.Id)
        If user Is Nothing Then
            MessageBox.Show("User not found.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error)
            Me.Close()
            Return
        End If

        txtEmail.Text = user.Email
        txtFullName.Text = user.FullName
    End Sub

    Private Sub BtnUpdate_Click(sender As Object, e As EventArgs)
        lblError.Text = ""

        Dim errors As New List(Of String)

        Dim email As String = txtEmail.Text.Trim()
        Dim fullName As String = txtFullName.Text.Trim()
        Dim password As String = txtPassword.Text

        errors.AddRange(ValidationHelper.ValidateEmail(email))
        errors.AddRange(ValidationHelper.ValidateFullName(fullName))

        If Not String.IsNullOrWhiteSpace(password) Then
            errors.AddRange(ValidationHelper.ValidatePassword(password))
        End If

        If errors.Count > 0 Then
            lblError.Text = String.Join(vbCrLf, errors)
            Return
        End If

        user.Email = email
        user.FullName = fullName

        Try
            UserRepository.Update(user)

            If Not String.IsNullOrWhiteSpace(password) Then
                UserRepository.UpdatePassword(user.Id, UserRepository.HashPassword(password))
            End If

            SessionManager.CurrentUser.FullName = user.FullName
            SessionManager.CurrentUser.Email = user.Email

            MessageBox.Show("Profile updated successfully!", "Success", MessageBoxButtons.OK, MessageBoxIcon.Information)
            Me.Close()
        Catch ex As Exception
            lblError.Text = "An error occurred: " & ex.Message
        End Try
    End Sub

    Private Sub BtnCancel_Click(sender As Object, e As EventArgs)
        Me.Close()
    End Sub
End Class

        Me.Text = "Airline Management System"

        Me.StartPosition = FormStartPosition.CenterScreen
        Me.WindowState = FormWindowState.Maximized

        menuStrip = New MenuStrip()

        flightsToolStripMenuItem = New ToolStripMenuItem("Flights")
        AddHandler flightsToolStripMenuItem.Click, AddressOf FlightsToolStripMenuItem_Click

        bookFlightToolStripMenuItem = New ToolStripMenuItem("Book Flight")
        AddHandler bookFlightToolStripMenuItem.Click, AddressOf BookFlightToolStripMenuItem_Click

        myBookingsToolStripMenuItem = New ToolStripMenuItem("My Bookings")
        AddHandler myBookingsToolStripMenuItem.Click, AddressOf MyBookingsToolStripMenuItem_Click

        ' Admin menu
        adminMenu = New ToolStripMenuItem("Admin")
        createFlightToolStripMenuItem = New ToolStripMenuItem("Create Flight")
        AddHandler createFlightToolStripMenuItem.Click, AddressOf CreateFlightToolStripMenuItem_Click
        adminMenu.DropDownItems.Add(createFlightToolStripMenuItem)

        customersToolStripMenuItem = New ToolStripMenuItem("Customers")
        AddHandler customersToolStripMenuItem.Click, AddressOf CustomersToolStripMenuItem_Click
        adminMenu.DropDownItems.Add(customersToolStripMenuItem)

        masterBookingsToolStripMenuItem = New ToolStripMenuItem("Master Bookings")
        AddHandler masterBookingsToolStripMenuItem.Click, AddressOf MasterBookingsToolStripMenuItem_Click
        adminMenu.DropDownItems.Add(masterBookingsToolStripMenuItem)

        menuStrip.Items.Add(flightsToolStripMenuItem)
        menuStrip.Items.Add(bookFlightToolStripMenuItem)
        menuStrip.Items.Add(myBookingsToolStripMenuItem)
        menuStrip.Items.Add(profileToolStripMenuItem)
        menuStrip.Items.Add(adminMenu)
        menuStrip.Items.Add(logoutToolStripMenuItem)

        Me.MainMenuStrip = menuStrip
        Me.Controls.Add(menuStrip)

        ' Welcome label
        lblWelcome = New Label()
        lblWelcome.Font = New Font("Arial", 12, FontStyle.Bold)
        lblWelcome.AutoSize = True
        lblWelcome.Location = New Point(20, 40)
        Me.Controls.Add(lblWelcome)
    End Sub

    Private Sub CheckSession()
        SessionManager.CheckTimeout()
        If Not SessionManager.IsAuthenticated() Then
            Me.Hide()
            Dim loginForm As New LoginForm()
            loginForm.Show()
            Me.Close()
        End If
    End Sub

    Private Sub UpdateMenu()
        If SessionManager.IsAuthenticated() Then
            lblWelcome.Text = "Welcome, " & SessionManager.CurrentUser.FullName & "!"
            adminMenu.Visible = SessionManager.IsAdmin()
        End If
    End Sub

    Private Sub FlightsToolStripMenuItem_Click(sender As Object, e As EventArgs)
        Dim flightsForm As New FlightsForm()
        flightsForm.ShowDialog()
    End Sub

    Private Sub BookFlightToolStripMenuItem_Click(sender As Object, e As EventArgs)
        ' Show flights form first to select a flight
        Dim flightsForm As New FlightsForm()
        flightsForm.ShowDialog()
    End Sub

    Private Sub MyBookingsToolStripMenuItem_Click(sender As Object, e As EventArgs)
        Dim bookingsForm As New MyBookingsForm()
        bookingsForm.ShowDialog()
    End Sub

    Private Sub ProfileToolStripMenuItem_Click(sender As Object, e As EventArgs)
        Dim profileForm As New ProfileForm()
        profileForm.ShowDialog()
    End Sub

    Private Sub CreateFlightToolStripMenuItem_Click(sender As Object, e As EventArgs)
        Dim createForm As New CreateFlightForm()
        createForm.ShowDialog()
    End Sub

    Private Sub CustomersToolStripMenuItem_Click(sender As Object, e As EventArgs)
        Dim customersForm As New CustomersForm()
        customersForm.ShowDialog()
    End Sub

    Private Sub MasterBookingsToolStripMenuItem_Click(sender As Object, e As EventArgs)
        Dim masterForm As New MasterBookingsForm()
        masterForm.ShowDialog()
    End Sub

    Private Sub LogoutToolStripMenuItem_Click(sender As Object, e As EventArgs)
        SessionManager.Logout()
        Me.Hide()
        Dim loginForm As New LoginForm()
        loginForm.Show()
        Me.Close()
    End Sub
End Class
